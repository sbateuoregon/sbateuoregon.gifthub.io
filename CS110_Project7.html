# prompt: what is the ratio of smokers and the relation to lung cancer

# Calculate the number of smokers and non-smokers
smoker_counts = df['SMOKING'].value_counts().reset_index()
smoker_counts.columns = ['SMOKING', 'Count']

# Create a bar plot to show the distribution of smokers vs. non-smokers
plt.figure(figsize=(6, 4))
sns.barplot(x='SMOKING', y='Count', data=smoker_counts)
plt.title('Distribution of Smokers (2) vs. Non-Smokers (1)')
plt.xlabel('Smoking Status (1: Non-Smoker, 2: Smoker)')
plt.ylabel('Number of Individuals')
plt.xticks(ticks=[0, 1], labels=['Non-Smoker (1)', 'Smoker (2)'])
plt.show()

# Analyze the relationship between smoking and lung cancer
# Group by smoking status and count the number of people with and without lung cancer
lung_cancer_by_smoking = df.groupby('SMOKING')['LUNG_CANCER'].value_counts().unstack().fillna(0)
lung_cancer_by_smoking.columns = ['No Lung Cancer (NO)', 'Lung Cancer (YES)']

# Convert to percentages for easier comparison
lung_cancer_by_smoking_percentage = lung_cancer_by_smoking.div(lung_cancer_by_smoking.sum(axis=1), axis=0) * 100

print("\nLung Cancer Distribution by Smoking Status:")
print(lung_cancer_by_smoking)

print("\nPercentage of Lung Cancer by Smoking Status:")
print(lung_cancer_by_smoking_percentage)

# Create a grouped bar chart to visualize the relationship
lung_cancer_by_smoking.plot(kind='bar', figsize=(8, 6))
plt.title('Lung Cancer Cases by Smoking Status')
plt.xlabel('Smoking Status (1: Non-Smoker, 2: Smoker)')
plt.ylabel('Number of Individuals')
plt.xticks(ticks=[0, 1], labels=['Non-Smoker (1)', 'Smoker (2)'], rotation=0)
plt.legend(title='Lung Cancer')
plt.show()
